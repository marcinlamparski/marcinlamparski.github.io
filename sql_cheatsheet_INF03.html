INF.03 Technikum Informatyczne
DDL — Definiowanie struktury bazy
-- Tworzenie tabeli CREATE TABLE uczniowie ( id INT PRIMARY KEY AUTO_INCREMENT, imie VARCHAR(50) NOT NULL, nazwisko VARCHAR(80) NOT NULL, data_ur DATE, klasa_id INT, FOREIGN KEY (klasa_id) REFERENCES klasy(id) );
-- Modyfikacja tabeli ALTER TABLE uczniowie ADD COLUMN email VARCHAR(100); ALTER TABLE uczniowie MODIFY COLUMN imie VARCHAR(80); ALTER TABLE uczniowie DROP COLUMN email; ALTER TABLE uczniowie RENAME TO studenci;
-- Usuwanie DROP TABLE uczniowie; -- usuwa tabelę DROP DATABASE szkola; -- usuwa bazę TRUNCATE TABLE logi; -- czyści dane
-- Tworzenie bazy i indeksu CREATE DATABASE szkola; CREATE INDEX idx_nazwisko ON uczniowie (nazwisko);
DML — Manipulowanie danymi
-- Wstawianie danych INSERT INTO uczniowie (imie, nazwisko, klasa_id) VALUES ('Anna', 'Kowalska', 3); -- Wiele rekordów naraz INSERT INTO uczniowie (imie, nazwisko) VALUES ('Jan', 'Nowak'), ('Ewa', 'Wiśniewska');
-- Aktualizacja danych UPDATE uczniowie SET klasa_id = 4, email = 'a@szk.pl' WHERE id = 7; -- ⚠ Bez WHERE – zmienia WSZYSTKIE wiersze!
-- Usuwanie danych DELETE FROM uczniowie WHERE id = 7; -- ⚠ Bez WHERE – usuwa WSZYSTKIE wiersze!
-- Pełne zapytanie SELECT SELECT kolumna1, kolumna2 FROM tabela WHERE warunek GROUP BY kolumna HAVING warunek_agregatu ORDER BY kolumna DESC LIMIT 10;
-- Kolejność wykonania (≠ pisania!) FROM → WHERE → GROUP BY → HAVING → SELECT → ORDER BY → LIMIT

SELECT — Klauzule i filtrowanie
-- Podstawowe SELECT * FROM uczniowie; SELECT DISTINCT klasa_id FROM uczniowie; SELECT imie AS imię_ucznia FROM uczniowie;
-- WHERE z operatorami WHERE wiek > 18 AND klasa = '3IT' WHERE wiek BETWEEN 16 AND 19 WHERE klasa IN ('2IT', '3IT', '4IT') WHERE email IS NULL WHERE email IS NOT NULL WHERE nazwisko LIKE 'Ko%' -- % = dowolny ciąg WHERE nazwisko LIKE '_owak' -- _ = 1 znak
-- GROUP BY + HAVING SELECT klasa_id, COUNT(*) AS liczba FROM uczniowie GROUP BY klasa_id HAVING COUNT(*) > 20; -- HAVING filtruje wyniki GROUP BY -- WHERE filtruje wiersze PRZED grupowaniem
-- ORDER BY + LIMIT ORDER BY nazwisko ASC -- rosnąco (domyślne) ORDER BY srednia DESC -- malejąco ORDER BY nazwisko, imie -- wielopoziomowe LIMIT 5 -- pierwsze 5 wyników LIMIT 10 OFFSET 20 -- strona 3 (po 10)

Podzapytania (Subqueries)
-- W WHERE SELECT * FROM uczniowie WHERE klasa_id = ( SELECT id FROM klasy WHERE nazwa = '3IT' );
-- Z IN (wiele wyników) SELECT * FROM uczniowie WHERE klasa_id IN ( SELECT id FROM klasy WHERE profil = 'informatyczny' );
-- W SELECT (skalarne) SELECT imie, (SELECT AVG(ocena) FROM oceny WHERE uczen_id = u.id) AS srednia FROM uczniowie u;
-- EXISTS SELECT * FROM uczniowie u WHERE EXISTS ( SELECT 1 FROM oceny o WHERE o.uczen_id = u.id AND o.ocena = 5 );
-- W FROM (tabela pochodna) SELECT * FROM ( SELECT klasa_id, AVG(ocena) avg_oc FROM oceny GROUP BY klasa_id ) AS srednie WHERE avg_oc > 4;

⚠ Pułapki egzaminacyjne
NULL – błędy porównania
WHERE x = NULL ❌ Zawsze fałsz!
WHERE x IS NULL ✓ Poprawnie
WHERE vs HAVING
WHERE Filtruje wiersze PRZED grupowaniem, nie używa funkcji agg.
HAVING Filtruje grupy PO GROUP BY, może używać COUNT, SUM...
DELETE vs TRUNCATE
DELETE Można z WHERE, wolniejsze, loguje zmiany
TRUNCATE Bez WHERE, szybsze, resetuje AUTO_INCREMENT
Kolejność klauzul SELECT
SELECT → FROM → WHERE → GROUP BY → HAVING → ORDER BY → LIMIT
Kompleksowy przykład — schemat bazy szkoły
-- Tworzenie schematu CREATE TABLE klasy ( id INT PRIMARY KEY AUTO_INCREMENT, nazwa VARCHAR(10) NOT NULL UNIQUE, profil VARCHAR(50) ); CREATE TABLE uczniowie ( id INT PRIMARY KEY AUTO_INCREMENT, imie VARCHAR(50) NOT NULL, nazwisko VARCHAR(80) NOT NULL, klasa_id INT, FOREIGN KEY (klasa_id) REFERENCES klasy(id) ); CREATE TABLE oceny ( id INT PRIMARY KEY AUTO_INCREMENT, uczen_id INT NOT NULL, przedmiot VARCHAR(60), ocena TINYINT CHECK (ocena BETWEEN 1 AND 6), data_wystawienia DATE DEFAULT (CURDATE()), FOREIGN KEY (uczen_id) REFERENCES uczniowie(id) );
-- Raport: średnia ocen każdego ucznia SELECT u.imie, u.nazwisko, k.nazwa AS klasa, COUNT(o.id) AS liczba_ocen, ROUND(AVG(o.ocena), 2) AS srednia, MIN(o.ocena) AS najnizsza, MAX(o.ocena) AS najwyzsza FROM uczniowie u JOIN klasy k ON u.klasa_id = k.id LEFT JOIN oceny o ON o.uczen_id = u.id GROUP BY u.id, u.imie, u.nazwisko, k.nazwa HAVING AVG(o.ocena) >= 4.0 ORDER BY srednia DESC LIMIT 10;
-- Widok: celujący uczniowie CREATE VIEW v_celujacy AS SELECT u.imie, u.nazwisko, k.nazwa, ROUND(AVG(o.ocena),2) AS sr FROM uczniowie u JOIN klasy k ON u.klasa_id = k.id JOIN oceny o ON o.uczen_id = u.id GROUP BY u.id, k.nazwa HAVING sr >= 5; -- Aktualizacja z podzapytaniem UPDATE uczniowie SET klasa_id = (SELECT id FROM klasy WHERE nazwa='4IT') WHERE id IN (SELECT uczen_id FROM v_celujacy);
